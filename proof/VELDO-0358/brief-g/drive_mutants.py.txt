import importlib.util as u, json, subprocess, sys, tempfile, difflib
from pathlib import Path
root = Path(sys.argv[1])
s = u.spec_from_file_location('t', root / 'scripts/check_teeth_mutations.py'); m = u.module_from_spec(s); s.loader.exec_module(m)
cases = [c for c in m.cases() if c['finding'] in (358, 313)]
def worker(case, mutant=None):
    cmd = [sys.executable, str(root / 'scripts/check_teeth_mutations.py'), '--worker', case['name']]
    if mutant: cmd += ['--mutant', str(mutant)]
    p = subprocess.run(cmd, capture_output=True, text=True, cwd=str(root))
    try:
        return p.returncode, json.loads(p.stdout.strip().splitlines()[-1]), p.stdout
    except Exception:
        return p.returncode, None, p.stdout[-2000:] + p.stderr[-2000:]
only = sys.argv[2:]
for case in cases:
    if only and case['name'] not in only: continue
    print('=' * 100); print('CASE', case['name'], 'targets', case['rows'])
    for mode in ('baseline', 'noop', 'mutant'):
        d = tempfile.mkdtemp(prefix='v358-mut-')
        info = m.materialize(case, mode, d)
        if mode == 'mutant':
            before = info['source'].read_text().splitlines(); after = Path(info['mutant']).read_text().splitlines() if Path(info['mutant']).is_file() else []
            print('DIFF (proves the mutant landed):'); print('\n'.join(difflib.unified_diff(before, after, 'subject', 'mutant', n=0, lineterm='')))
        rc, res, raw = worker(case, info['mutant'])
        if res is None:
            print(mode, 'rc', rc, 'NO JSON:', raw); continue
        print(mode, 'rc', rc, 'rows', res['count'], 'failed', res['failed_rows'], 'targets', res['targets'])
        for row, detail in res['failed_details']:
            print('   detail', row, detail[:700])
